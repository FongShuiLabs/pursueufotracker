name: Poll war.gov for new PURSUE drops

# Polls the official war.gov UAP CSV every 30 minutes during US weekday business
# hours (when historical drops have happened) and hourly otherwise. Opens an
# issue AND pushes an ntfy alert when a new drop is detected; commits the
# updated state file on every successful poll.

on:
  schedule:
    # Every 30 min, weekdays 13:00-22:00 UTC (9am-6pm ET)
    - cron: '*/30 13-22 * * 1-5'
    # Hourly off-hours, 7 days a week
    - cron: '0 0-12,23 * * *'
  workflow_dispatch:

permissions:
  contents: write
  issues: write

concurrency:
  group: poll-wargov
  cancel-in-progress: false

jobs:
  poll:
    runs-on: ubuntu-latest
    steps:
      # Node 24 runtimes. checkout@v4 / setup-python@v5 are node20, which GitHub
      # deprecated (Sept 2025) and now force-runs on node24, emitting a warning
      # on every run. Verified from each action's manifest 2026-07-23:
      # checkout v5/v6/v7 = node24, setup-python v6/v7 = node24.
      # v7 pinned (both released 2026-07-20). The 2026-07-20 "allow-unsafe-pr-checkout"
      # breaking change only affects pull_request_target, which this workflow
      # does not use (schedule + workflow_dispatch only).
      - name: Checkout
        uses: actions/checkout@v7
        with:
          fetch-depth: 1

      - name: Set up Python
        uses: actions/setup-python@v7
        with:
          python-version: '3.12'
          cache: 'pip'

      - name: Install poller deps
        run: pip install curl_cffi

      - name: Poll war.gov
        id: poll
        continue-on-error: true
        run: python -m pipeline.poll_wargov

      - name: Commit poll state
        if: steps.poll.outcome == 'success' || steps.poll.outputs.new_drop == 'true'
        run: |
          git config user.name "pursueufotracker-bot"
          git config user.email "bot@pursueufotracker.com"
          git add data/poll-state.json
          if git diff --cached --quiet; then
            echo "No state change to commit."
          else
            if [ "${{ steps.poll.outputs.new_drop }}" = "true" ]; then
              git commit -m "[NEW DROP] war.gov CSV changed: ${{ steps.poll.outputs.row_count }} rows (${{ steps.poll.outputs.delta }} delta)"
            else
              git commit -m "poll: war.gov unchanged ($(date -u +%FT%TZ)) [CF-Pages-Skip]"
            fi
            # Retry with rebase: a bare `git push` fails non-fast-forward whenever
            # a human pushes between this job's checkout and its push. That is the
            # ONLY cause of workflow failure in 518 runs (run #462, 2026-07-17,
            # during a burst of operator commits). Detection never stopped - only
            # the state commit - but a red X on the drop poller is noise we do not
            # want to learn to ignore. poll-state.json is bot-owned, so rebasing
            # onto origin/main is always safe here.
            for attempt in 1 2 3; do
              if git push; then
                echo "state pushed (attempt $attempt)"; break
              fi
              if [ "$attempt" = "3" ]; then
                echo "::warning::Could not push poll state after 3 attempts. Detection still ran; the next poll will re-commit."
                exit 0
              fi
              echo "push rejected - rebasing onto origin/main and retrying"
              git pull --rebase --autostash origin main || exit 0
              sleep 3
            done
          fi

      # github-script@v7 is also node20 (verified 2026-07-23). It did NOT appear in
      # the deprecation warning only because this step runs solely when a drop
      # fires - i.e. it would have surfaced during the highest-stakes run of the
      # cycle. v9 is node24.
      - name: Open NEW DROP issue
        if: steps.poll.outputs.new_drop == 'true'
        uses: actions/github-script@v9
        with:
          script: |
            const rowCount = '${{ steps.poll.outputs.row_count }}';
            const delta = '${{ steps.poll.outputs.delta }}';
            const sha = '${{ steps.poll.outputs.sha256 }}';
            const body = [
              '## war.gov uap-data.csv has changed',
              '',
              `- New row count: **${rowCount}**`,
              `- Delta: **${delta}**`,
              `- New CSV SHA-256: \`${sha}\``,
              `- Detected: ${new Date().toISOString()}`,
              '',
              '### Follow DROP_REACTION.md top-to-bottom. Short version:',
              '1. `git pull`',
              '2. `python -m pipeline.preflight pre-ingest`  (fetches a verified-fresh CSV; never skip)',
              '3. `python -m pipeline.run all`',
              '4. `python -m pipeline.preflight post-ingest`  (then RECONCILE any count-drift WARNings)',
              '5. Append the new drop to `data/drops.json`, then `python -m pipeline.run build-drops`',
              '6. `python -m pipeline.preflight pre-push`, then push',
              '7. `python -m pipeline.drop_check --mark` to clear the guard',
              '',
              'Verify the id-level diff before ANY public claim (Hard Rule #7).',
              'Triggered automatically by `.github/workflows/poll-wargov.yml`.',
            ].join('\n');
            await github.rest.issues.create({
              owner: context.repo.owner,
              repo: context.repo.repo,
              title: `[NEW DROP] war.gov UAP CSV changed (${delta} new rows, ${rowCount} total)`,
              body,
              labels: ['new-drop', 'automated'],
            });

      # Values are passed via env (not inlined into the shell) so a weird CSV
      # value can never be interpreted as a command. Non-fatal by design: a
      # flaky notification must never break drop detection.
      - name: Push ntfy drop alert
        if: steps.poll.outputs.new_drop == 'true'
        continue-on-error: true
        env:
          NTFY_TOPIC: ${{ secrets.NTFY_TOPIC }}
          ROWS: ${{ steps.poll.outputs.row_count }}
          DELTA: ${{ steps.poll.outputs.delta }}
          SHA: ${{ steps.poll.outputs.sha256 }}
          REPO: ${{ github.repository }}
        run: |
          if [ -z "$NTFY_TOPIC" ]; then
            echo "NTFY_TOPIC secret not set - skipping push. Poller unaffected."
            exit 0
          fi
          MSG=$(printf '%s\n' \
            "**war.gov just published a new PURSUE drop.**" \
            "" \
            "* Rows: **$ROWS**  (delta $DELTA)" \
            "* CSV SHA-256: \`$SHA\`" \
            "* Detected: $(date -u +%FT%TZ)" \
            "" \
            "**Run DROP_REACTION.md:**" \
            "1. preflight pre-ingest" \
            "2. pipeline.run all" \
            "3. preflight post-ingest (reconcile count drift)" \
            "4. append data/drops.json + build-drops" \
            "5. preflight pre-push, then push" \
            "" \
            "Verify the id-level diff before any public claim.")
          if curl -sS --max-time 20 \
              -H "Title: NEW DROP - war.gov UAP CSV changed" \
              -H "Tags: rotating_light,satellite" \
              -H "Priority: high" \
              -H "Markdown: yes" \
              -H "Click: https://github.com/$REPO/issues" \
              -d "$MSG" \
              "https://ntfy.sh/$NTFY_TOPIC" ; then
            echo "ntfy alert sent."
          else
            echo "::warning::ntfy push failed (non-fatal). The NEW DROP issue was still created."
          fi

      - name: Fail if poll errored
        if: steps.poll.outcome == 'failure' && steps.poll.outputs.new_drop != 'true'
        run: |
          echo "::error::Poll script failed (network/parse error). Investigate logs above."
          exit 1
